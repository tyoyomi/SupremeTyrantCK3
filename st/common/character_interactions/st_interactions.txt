### Grant / revoke (when rule allows targets)
st_grant_supreme_tyrant_interaction = {
	category = interaction_category_friendly
	icon = st_grant_supreme_tyrant
	interface_priority = 58
	common_interaction = yes

	desc = st_grant_supreme_tyrant_interaction_desc
	auto_accept = yes

	is_shown = {
		has_game_rule = st_rule_granting_player_and_targets
		scope:actor = {
			is_alive = yes
			st_actor_may_bestow_by_rule = yes
		}
		scope:recipient = {
			is_alive = yes
			is_adult = yes
			NOT = { has_trait = st_supreme_tyrant }
			OR = {
				is_child_of = scope:actor
				is_grandchild_of = scope:actor
			}
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = {
			custom_tooltip = {
				text = st_tt_can_receive_tyrant
				is_adult = yes
				NOT = { has_trait = st_supreme_tyrant }
			}
		}
	}

	on_accept = {
		scope:recipient = {
			st_embrace_supreme_tyrant_effect = yes
		}
	}

	ai_will_do = { base = 0 }
}

				is_vassal_of = scope:actor
			}
		}
	}

	on_accept = {
		st_force_oath_of_fear_effect = yes
	}

	ai_accept = {
		base = 10
		st_fear_ai_accept_modifier = yes

			scope:actor = { dread < 40 }
		}
	}

	ai_will_do = { base = 0 }
